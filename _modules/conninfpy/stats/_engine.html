"""Shared permutation machinery used by both the t-test and GLM pipelines.

Stratified (within-block) permutation, max-statistic extraction, collection
of per-permutation results, empirical p-values with the +1 Phipson–Smyth
correction, BH-FDR, and the worker-process / core-count helpers.
:mod:`conninfpy.stats.pairwise` and :mod:`conninfpy.stats.glm` build their
permutation loops from these pieces.
"""

from __future__ import annotations

import multiprocessing
import os
from typing import Any, Dict, List, Tuple, Union

import numpy as np

def _twosample_se2(var1, n1, var2, n2, var_equal=False):
    """Squared standard error of ``mean2 - mean1``.

    The one place where the two two-sample t statistics differ; the observed
    statistic, every permutation and the available-case path use it, so the
    observed t and its null are always of the same kind.

    - Welch (``var_equal=False``): ``var1/n1 + var2/n2``.
    - Student (``var_equal=True``): pooled variance
      ``((n1-1) var1 + (n2-1) var2) / (n1+n2-2)`` times ``1/n1 + 1/n2``.

    ``n1``/``n2`` may be per-edge arrays (available-case analysis).
    """
    if var_equal:
        pooled = ((n1 - 1) * var1 + (n2 - 1) * var2) / (n1 + n2 - 2)
        return pooled * (1.0 / n1 + 1.0 / n2)
    return var1 / n1 + var2 / n2


def _encode_strata(strata: Any) -> npt.NDArray[np.int_]:
    """Map an arbitrary stratum-label sequence to contiguous integer codes."""
    arr = np.asarray(strata)
    _, codes = np.unique(arr, return_inverse=True)
    return codes.astype(np.int_)


def _stratified_perm(
    strata: npt.NDArray[np.int_], rng: np.random.RandomState
) -> npt.NDArray[np.int_]:
    """Return a permutation index honoring exchangeability blocks.

    Within each unique stratum value, indices are independently
    permuted; across strata, indices do not move. This implements
    within-block exchangeability used by Freedman-Lane permutation
    on multi-site or otherwise blocked designs.

    Parameters
    ----------
    strata : ndarray of shape (n,)
        Integer stratum codes (use :func:`_encode_strata` first).
    rng : numpy.random.RandomState
        Already-seeded RNG.

    Returns
    -------
    perm : ndarray of shape (n,)
        Permutation index such that ``perm[i]`` is the new position of
        the data point originally at ``i``. Equivalent: for any array
        ``a``, ``a[perm]`` is the stratum-respecting permutation of ``a``.
    """
    n = strata.shape[0]
    perm = np.arange(n)
    for s in np.unique(strata):
        idx = np.where(strata == s)[0]
        if idx.size > 1:
            perm[idx] = rng.permutation(idx)
    return perm


def _stratified_choice_n1(
    strata: npt.NDArray[np.int_],
    n1_per_stratum: npt.NDArray[np.int_],
    rng: np.random.RandomState,
) -> npt.NDArray[np.float64]:
    """Length-n binary mask honoring per-stratum group-1 counts.

    For the two-sample fast permutation path under exchangeability
    blocking: within each stratum ``s``, sample exactly
    ``n1_per_stratum[s]`` indices to label as group 1 (the rest are
    group 2). Across strata, the per-stratum group totals are held
    fixed.

    Parameters
    ----------
    strata : ndarray of shape (n,)
        Integer stratum codes.
    n1_per_stratum : ndarray of shape (n_unique_strata,)
        Group-1 count per stratum (computed once from the observed
        group labels and reused across permutations).
    rng : numpy.random.RandomState
    """
    n = strata.shape[0]
    g = np.zeros(n, dtype=np.float64)
    for s_code, n1_s in enumerate(n1_per_stratum):
        if n1_s <= 0:
            continue
        idx = np.where(strata == s_code)[0]
        chosen = rng.choice(idx, size=int(n1_s), replace=False)
        g[chosen] = 1.0
    return g


def _extract_max_stats(
    stat_dict: Dict[str, npt.NDArray], reference_shape: Tuple[int, ...]
) -> Dict[str, Union[float, npt.NDArray[np.float64]]]:
    """
    Extract maximum statistics from a stat dictionary.

    Handles both 2D matrices and multi-parameter 3D arrays.
    Key-agnostic: works with any dictionary keys (e.g., 'negative'/'positive'
    for t-test pipeline, 'positive'/'negative' for GLM pipeline).

    Parameters
    ----------
    stat_dict : dict
        Dictionary with statistic arrays (any string keys).
    reference_shape : tuple
        Shape of a single sample for determining dimensionality.

    Returns
    -------
    dict
        Dictionary with maximum statistics for each key.
    """
    result: Dict[str, Any] = {}
    for key, arr in stat_dict.items():
        if arr.shape == reference_shape:
            result[key] = np.max(arr).astype(np.float64)
        else:
            # Multi-parameter case: max over spatial dims, keep param dim
            spatial_axes = tuple(range(arr.ndim - 1))
            result[key] = np.max(arr, axis=spatial_axes).astype(np.float64)
    if set(result.keys()) == {"positive", "negative"}:
        return make_tail_result(result["positive"], result["negative"])
    return result


def _collect_results_to_arrays(
    results: List[Dict[str, Any]], n_permutations: int
) -> Dict[str, npt.NDArray[np.float64]]:
    """
    Efficiently collect permutation results into numpy arrays.

    Uses pre-allocation and direct indexing instead of Python loops.

    Parameters
    ----------
    results : list of dict
        List of permutation result dictionaries.
    n_permutations : int
        Number of permutations.

    Returns
    -------
    dict
        Dictionary with arrays of shape (n_permutations,) or (n_permutations, n_params).
    """
    group_keys = list(results[0].keys())
    first_val = results[0][group_keys[0]]
    output_shape = first_val.shape if hasattr(first_val, "shape") else ()

    t_maxes_dict = {
        key: np.empty((n_permutations, *output_shape), dtype=np.float64)
        for key in group_keys
    }

    for key in group_keys:
        for i, perm_dict in enumerate(results):
            t_maxes_dict[key][i] = perm_dict[key]

    if set(t_maxes_dict.keys()) == {"positive", "negative"}:
        return make_tail_result(t_maxes_dict["positive"], t_maxes_dict["negative"])
    return t_maxes_dict


def _is_worker_process() -> bool:
    """Check if running inside a multiprocessing worker.

    Returns True when the current process was spawned by a Pool,
    preventing nested Pool creation which causes deadlocks and
    unintentional process multiplication.
    """
    proc_name = multiprocessing.current_process().name
    return (
        proc_name != "MainProcess"
        or "PoolWorker" in proc_name
        or "SpawnPoolWorker" in proc_name
        or "ForkPoolWorker" in proc_name
    )

def get_available_cores():
    """Usable logical CPUs, including affinity, container quota and soft limits."""
    count = _cpu_count()
    process_count = getattr(os, "process_cpu_count", None)
    if process_count is not None:
        process_limit = process_count()
        if process_limit is not None:
            count = min(count, process_limit)
    return max(1, count)




def _check_finite_statistics(
    observed: npt.NDArray, null_dist: npt.NDArray, key: str
) -> None:
    """Refuse to turn NaN statistics into p-values.

    ``x >= NaN`` is always False, so a NaN in the null maxima would give every
    edge the smallest possible p-value. Inputs are checked on entry; this
    catches NaN created inside a computation.
    """
    if np.isnan(null_dist).any() or np.isnan(observed).any():
        raise RuntimeError(
            f"NaN statistics in the {key!r} tail (observed: "
            f"{int(np.isnan(observed).sum())} values, permutation maxima: "
            f"{int(np.isnan(null_dist).sum())} of {null_dist.shape[0]}); "
            "p-values would be meaningless. Check the input for edges that "
            "are constant across all subjects."
        )


def _compute_p_values_from_null(
    emp_t_dict: Dict[str, npt.NDArray],
    max_null_dict: Dict[str, npt.NDArray],
    *,
    return_counts: bool = False,
) -> Any:
    """
    Compute max-statistic p-values from empirical statistics and a null.

    Ties are counted in the conservative direction (``null >= observed``)
    and the Phipson-Smyth +1 correction is applied:
    ``p = (count + 1) / (B + 1)``. This keeps an all-zero observed map
    against an all-zero null at ``p = 1`` rather than ``1 / (B + 1)``.

    Parameters
    ----------
    emp_t_dict : dict
        Dictionary with empirical t-statistic arrays.
    max_null_dict : dict
        Dictionary with null distribution arrays.
    return_counts : bool, default False
        Also return the raw per-cell exceedance counts (``null >= observed``,
        one 0/1 indicator per draw, summed). They are genuinely computed here,
        never reconstructed from the p-values. The default return value is
        unchanged.

    Returns
    -------
    dict or (dict, dict)
        Dictionary with p-value arrays for each direction, and, with
        ``return_counts=True``, a second dictionary of int64 count arrays
        (same shape as the p-value arrays).
    """
    keys = list(emp_t_dict.keys())
    p_values = {}
    counts_out: Dict[str, npt.NDArray[np.int64]] = {}

    is_2d = len(emp_t_dict[keys[0]].shape) == 2

    for key in keys:
        emp_t = emp_t_dict[key]
        null_dist = max_null_dict[key]
        _check_finite_statistics(emp_t, null_dist, key)

        n_perm = null_dist.shape[0]
        if is_2d:
            if null_dist.ndim == 1:
                # Shape: (N, N) vs (n_permutations,). Count null >= observed
                # (ties included, identical to the broadcast comparison)
                # without materialising an (N, N, B) boolean array: sort the
                # null once, then the count is B minus the number of null
                # values strictly below the observed statistic.
                count = np.empty(emp_t.shape, dtype=np.int64)
                sorted_null = np.sort(null_dist)
                count[...] = n_perm - np.searchsorted(sorted_null, emp_t, side="left")
            else:
                # Shape: (N, N) vs (n_permutations, n_params) — the broadcast
                # contract of the base expression ``emp[..., None] <= null``
                # summed over the LAST (params) axis, without materialising
                # the (N, B, P) comparison: column k of the null (shape (B,))
                # right-aligns against emp (fusing with emp's last axis when
                # sizes match, exactly like the base broadcast); incompatible
                # axes raise ValueError from np.broadcast_shapes, the same
                # failure type as the base comparison. Singleton null axes
                # stretch; the resulting count may exceed B when B does not
                # feed the summed axis — kept verbatim from base.
                out_shape = np.broadcast_shapes(
                    emp_t.shape + (1,), null_dist.shape
                )
                count = np.zeros(out_shape[:-1], dtype=np.int64)
                emp_full = np.broadcast_to(emp_t[..., None], tuple(out_shape))
                for param in range(null_dist.shape[1]):
                    # Column k: the (B,)-slice at parameter k, kept on the B
                    # axis (second-to-last of out) so singletons and fused
                    # axes broadcast exactly as in the base comparison.
                    col_full = np.broadcast_to(
                        null_dist[:, param][:, None], tuple(out_shape)
                    )
                    count += emp_full[..., 0] <= col_full[..., 0]
        else:
            # Multi-param: (N, N, n_params) vs (n_permutations, n_params),
            # with the parameter-broadcast contract of the original
            # ``emp_t[..., np.newaxis] <= null_dist.swapaxes(0, 1)[None, None]``
            # preserved without materialising the (N, N, P, B) array:
            # a singleton parameter axis on either side stretches, mutually
            # incompatible parameter axes raise ValueError
            # (np.broadcast_shapes, the same failure type as the original
            # comparison), and a 1D null in the multi-param branch raises
            # AxisError exactly like the original swapaxes(0, 1).
            null_swapped = null_dist.swapaxes(0, 1)
            out_shape = np.broadcast_shapes(
                emp_t.shape + (1,), (1, 1) + null_swapped.shape
            )
            n_params = out_shape[-2]  # trailing axis of out_shape is B
            count_shape = out_shape[:-1]
            count = np.empty(count_shape, dtype=np.int64)
            # Observed values along the parameter axis of the original
            # broadcast: emp[..., None] spread over the full comparison shape
            # (a singleton emp parameter axis stretches, a full-size axis
            # maps index to index, fused axes behave exactly as in the
            # original expression) — the [:, param, 0] slice is the
            # comparison minus the B axis.
            emp_full = np.broadcast_to(emp_t[..., None], tuple(out_shape))
            for param in range(n_params):
                pn = 0 if null_dist.shape[1] == 1 else param
                sorted_col = np.sort(null_dist[:, pn])
                emp_values = emp_full[..., param, 0]
                count[..., param] = n_perm - np.searchsorted(
                    sorted_col, emp_values, side="left"
                )

        # +1 correction (Phipson & Smyth 2010): prevents p = 0 with finite perms
        p_values[key] = (count + 1.0) / (n_perm + 1.0)
        if return_counts:
            counts_out[key] = np.asarray(count, dtype=np.int64)

    if return_counts:
        return p_values, counts_out
    return p_values


def _bh_fdr_correction(
    p_values: npt.NDArray[np.float64],
) -> npt.NDArray[np.float64]:
    """
    Apply Benjamini-Hochberg FDR correction to a 1D array of p-values.

    Parameters
    ----------
    p_values : ndarray of shape (m,)
        Uncorrected p-values.

    Returns
    -------
    ndarray of shape (m,)
        BH-corrected p-values (adjusted so that thresholding at alpha
        controls FDR at level alpha).
    """
    m = len(p_values)
    if m == 0:
        return p_values.copy()

    # Sort p-values and track original indices
    sorted_idx = np.argsort(p_values)
    sorted_p = p_values[sorted_idx]

    # BH adjustment: p_adj[i] = p[i] * m / rank[i]
    ranks = np.arange(1, m + 1, dtype=np.float64)
    adjusted = sorted_p * m / ranks

    # Enforce monotonicity (step-up): working backwards,
    # each adjusted p-value must be <= the next one
    adjusted = np.minimum.accumulate(adjusted[::-1])[::-1]

    # Cap at 1.0
    adjusted = np.minimum(adjusted, 1.0)

    # Restore original order
    result = np.empty(m, dtype=np.float64)
    result[sorted_idx] = adjusted

    return result
